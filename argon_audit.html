""" SUGGESTINVEST - CORE ENGINE & STRATEGY MODULES Konsolidovaný kód pro kvantitativní audit AI Argon """ import numpy as np import pandas as pd def compute_atr(df, period=14): tr1 = df['high'] - df['low'] tr2 = (df['high'] - df['close'].shift(1)).abs() tr3 = (df['low'] - df['close'].shift(1)).abs() tr = pd.concat([tr1, tr2, tr3], axis=1).max(axis=1) return tr.rolling(window=period, min_periods=1).mean().bfill().clip(lower=0.85) def compute_adx(df, period=14): plus_dm = df['high'].diff() minus_dm = -df['low'].diff() plus_dm = np.where((plus_dm > minus_dm) & (plus_dm > 0), plus_dm, 0.0) minus_dm = np.where((minus_dm > plus_dm) & (minus_dm > 0), minus_dm, 0.0) tr = compute_atr(df, period) plus_di = 100 * (pd.Series(plus_dm, index=df.index).rolling(period).mean() / tr) minus_di = 100 * (pd.Series(minus_dm, index=df.index).rolling(period).mean() / tr) dx = (abs(plus_di - minus_di) / (plus_di + minus_di + 1e-9)) * 100 return dx.rolling(period).mean().fillna(0) # Fractional Kelly Formula def calculate_fractional_kelly(win_rate, win_loss_ratio, fraction=0.25): p = win_rate b = win_loss_ratio kelly_f = max(0.0, (p * (b + 1) - 1) / (b + 1e-9)) return kelly_f * fraction # Triple Barrier Sizing & Execution def execute_triple_barrier(entry_price, direction, atr, tp_mult=2.5, sl_mult=1.0): if direction == 'LONG': tp = entry_price + tp_mult * atr sl = entry_price - sl_mult * atr else: tp = entry_price - tp_mult * atr sl = entry_price + sl_mult * atr return round(tp, 2), round(sl, 2)
